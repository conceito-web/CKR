/* ---------- Carrinho ---------- */
function adicionar(id, tam){
  const ex = cart.find(i => i.id===id && i.tam===tam);
  if (ex) ex.qtd++; else cart.push({id, tam, qtd:1});
  renderCarrinho();
  toast("Adicionado ao carrinho ✔");
}
function totalCarrinho(){
  return cart.reduce((s,i)=> s + precoDe(PRODUTOS.find(p=>p.id===i.id), i.tam) * i.qtd, 0);
}
function renderCarrinho(){
  $("#contador").textContent = cart.reduce((s,i)=>s+i.qtd,0);
  $("#listaCarrinho").innerHTML = cart.length ? cart.map((i,idx) => {
    const p = PRODUTOS.find(x=>x.id===i.id);
    const t = TAMANHOS.find(x=>x.id===i.tam);
    return `<div class="item">
      <div><strong>${p.nome}</strong><br><small>${t.nome}</small>
        <div class="qtd"><button data-m="${idx}">−</button><span>${i.qtd}</span><button data-p="${idx}">+</button></div>
      </div>
      <div style="text-align:right"><strong>${brl(precoDe(p,i.tam)*i.qtd)}</strong><br>
        <button class="mini" data-r="${idx}" style="margin-top:8px">Remover</button></div>
    </div>`;
  }).join("") : `<p class="sub">Seu carrinho está vazio.</p>`;
  $("#totalCarrinho").textContent = brl(totalCarrinho());
  document.querySelectorAll("[data-m]").forEach(b => b.onclick = () => { const i=cart[b.dataset.m]; i.qtd--; if(i.qtd<=0) cart.splice(b.dataset.m,1); renderCarrinho(); });
  document.querySelectorAll("[data-p]").forEach(b => b.onclick = () => { cart[b.dataset.p].qtd++; renderCarrinho(); });
  document.querySelectorAll("[data-r]").forEach(b => b.onclick = () => { cart.splice(b.dataset.r,1); renderCarrinho(); });
  atualizarLinkZap();
}
function atualizarLinkZap(){
  const nome = $("#nomeCliente").value.trim();
  const obs = $("#obsCliente").value.trim();
  let msg = `Olá, ${CONFIG.loja}! Quero fazer este pedido:\n\n`;
  if(!cart.length) msg = `Olá, ${CONFIG.loja}! Gostaria de mais informações sobre os colchões.`;
  else {
    cart.forEach(i => {
      const p = PRODUTOS.find(x=>x.id===i.id), t = TAMANHOS.find(x=>x.id===i.tam);
      msg += `• ${i.qtd}x ${p.nome} (${t.nome}) — ${brl(precoDe(p,i.tam)*i.qtd)}\n`;
    });
    msg += `\nTotal: ${brl(totalCarrinho())}`;
    if(nome) msg += `\nNome: ${nome}`;
    if(obs) msg += `\nObs.: ${obs}`;
  }
  $("#finalizar").href = `https://wa.me/${CONFIG.whatsapp}?text=${encodeURIComponent(msg)}`;
}

/* ---------- Comparador ---------- */
function alternarComparar(id, chk){
  if (comparar.includes(id)) comparar = comparar.filter(x=>x!==id);
  else {
    if (comparar.length>=3){ chk.checked=false; toast("Máximo de 3 colchões"); return; }
    comparar.push(id);
  }
  atualizarComparador();
}
function atualizarComparador(){
  const bar = $("#compBar");
  bar.classList.toggle("on", comparar.length>0);
  $("#compTexto").textContent = `${comparar.length} selecionado(s)`;
  const box = $("#tabelaComp");
  if(!comparar.length){ box.innerHTML = `<p class="sub">Nenhum colchão selecionado ainda.</p>`; return; }
  const ps = comparar.map(id => PRODUTOS.find(p=>p.id===id));
  const linha = (rot, fn) => `<tr><th>${rot}</th>${ps.map(p=>`<td>${fn(p)}</td>`).join("")}</tr>`;
  const estrelas = n => "●".repeat(n) + "○".repeat(5-n);
  box.innerHTML = `<table>
    <tr><th></th>${ps.map(p=>`<td><strong>${p.nome}</strong></td>`).join("")}</tr>
    ${linha("A partir de", p=>brl(precoDe(p,"solteiro")))}
    ${linha("Firmeza", p=>estrelas(p.firmeza))}
    ${linha("Frescor", p=>estrelas(p.frescor))}
    ${linha("Durabilidade", p=>estrelas(p.durab))}
    ${linha("Destaque", p=>p.tag)}
    ${linha("", p=>`<button class="mini pri" data-abrir2="${p.id}">Ver / Comprar</button>`)}
  </table>`;
  box.querySelectorAll("[data-abrir2]").forEach(b => b.onclick = () => abrirModal(b.dataset.abrir2));
}

/* ---------- Quiz ---------- */
const PERGUNTAS = [
  { q:"Como você costuma dormir?", o:[["De barriga para cima","costas"],["De lado","lado"],["De barriga para baixo","barriga"]] },
  { q:"Qual firmeza você prefere?", o:[["Macio, tipo abraço","macio"],["Equilibrado","medio"],["Bem firme","firme"]] },
  { q:"Você sente calor à noite?", o:[["Sim, muito","calor"],["Um pouco","pouco"],["Não","nao"]] }
];
let respostas = [];
function quiz(){
  const box = $("#quizBox");
  const n = respostas.length;
  if(n < PERGUNTAS.length){
    const q = PERGUNTAS[n];
    box.innerHTML = `<div class="prog"><b style="width:${(n/PERGUNTAS.length)*100}%"></b></div>
      <div class="pergunta">${q.q}</div>
      <div class="opcoes">${q.o.map(([t,v])=>`<button data-v="${v}">${t}</button>`).join("")}</div>`;
    box.querySelectorAll("[data-v]").forEach(b => b.onclick = () => { respostas.push(b.dataset.v); quiz(); });
    return;
  }
  const [pos, fir, cal] = respostas;
  const alvoF = fir==="macio"?2 : fir==="firme"?5 : 3;
  const alvoC = cal==="calor"?5 : cal==="pouco"?3 : 1;
  const pontuados = PRODUTOS.map(p => {
    let s = 10 - Math.abs(p.firmeza-alvoF)*2 - Math.abs(p.frescor-alvoC);
    if(pos==="barriga" && p.firmeza>=3) s += 1;
    if(pos==="lado" && p.firmeza<=3) s += 1;
    if(pos==="costas" && p.firmeza>=3) s += 1;
    return {p, s};
  }).sort((a,b)=>b.s-a.s);
  const melhor = pontuados[0].p;
  box.innerHTML = `<div class="prog"><b style="width:100%"></b></div>
    <div class="resultado">
      <div class="pergunta">Nossa recomendação: ${melhor.nome} ${melhor.emoji}</div>
      <p style="color:var(--mut)">${melhor.desc}</p>
      <div class="opcoes">
        <button class="btn pri" id="verRec">Ver e comprar</button>
        <button id="refazer">Refazer o quiz</button>
      </div>
      <p style="color:var(--mut);font-size:.9rem">Outras boas opções: ${pontuados.slice(1,3).map(x=>x.p.nome).join(" e ")}.</p>
    </div>`;
  $("#verRec").onclick = () => abrirModal(melhor.id);
  $("#refazer").onclick = () => { respostas = []; quiz(); };
}
